% ---------------------------------------------------------------- OVERVIEW
\begin{frame}{The question}
\begin{itemize}
  \item Collective behaviors (innovations, mobilizations) spread as \emphr{adoption cascades} on social networks.
  \vspace{0.3cm}
  \item Most models use \emph{synthetic} topologies and treat agents as cognition-free relays \refn{1}.
  \vspace{0.3cm}
  \item Real networks are shaped by \emphr{multidimensional homophily} — similar people cluster \refn{2}.
\end{itemize}
\begin{center}
\fcolorbox{ndred}{white}{\parbox{1.0\textwidth}{
\centering
Conditional on \emph{fixed individual propensities},

Can the \emphr{shape of realistic networks alone} change the outcome? 

Can the structure itself be an \emphr{autonomous force},
deciding not just \emph{how much} spreads, but \emph{how} it spreads?}}
\end{center}
\end{frame}

% ================================================================ THE MODEL
\sectionslide{The model}

% ---- Rational choice  (point 1: separated IUL/MUR ; point 2: 3 rational snapshots)
\begin{frame}{The model — 1) Rational Choice}
\begin{itemize}
  \item The attractiveness of the innovation is its
        \emph{Intrinsic Utility Level} (IUL), $\Gamma\in[0,1]$.
  \item Each person has a \emph{Minimum Utility Requirement} (MUR), $q_i\in[0,1]$,
        representing their aversion to adopting the behavior.
\end{itemize}
\vspace{0.15cm}
\begin{center}
An agent adopts on its own iff\quad
\fcolorbox{ndred}{white}{$\;q_i \le \Gamma \;\implies\; a_i = 1\;$}
\end{center}
\vspace{0.25cm}
\begin{center}
\includegraphics[width=0.95\textwidth]{graph-rational-choice.png}
\end{center}
\end{frame}

% ---- Selective social influence (point 2: 3 social snapshots)
\begin{frame}{The model — 2) Selective Social Influence}
Adopt by peer contagion if $\tau_i \le \tilde{E}_i$ \refn{5,6} 
— but only those socially \emph{close} \refn{7} count:
\vspace{0.05cm}
\begin{equation*}
  \tilde{E}_i \equiv \frac{\sum_{j\neq i}\mathbf{X}_{ij}\,\tilde{a}_j}{\sum_{j\neq i}\mathbf{X}_{ij}},
  \qquad
  \tilde{a}_j = 1 \iff a_j = 1 \;\wedge\; U_{ij} \le \sigma\!\left(MSP - d_{ij}\right)
\end{equation*}
\begin{itemize}
  \item Where \emphr{MSP} (Maximum Social Proximity) sets how far influence reaches. 
\end{itemize}
\vspace{0.05cm}
\begin{center}
\includegraphics[width=0.9\textwidth]{graph-social-influence.png}
\end{center}
\end{frame}

\begin{frame}{Comprehensive simulation design}
Hybrid social contagion via \texttt{netdiffuseR} \refn{9}, over plausible \emphr{GSS} vs.\
degree-preserving randomized \emphr{GSS-DP} topologies.
\vspace{0.2cm}
\begin{itemize}
  \item \textbf{IUL} ($\Gamma$): 41 levels $[0,1]$. \quad \textbf{MSP} ($h$): 13 levels $[0,1]$.
  \item \textbf{Thresholds} $\tau_i\sim\mathcal{N}(\mu,\sigma)$: $\mu\in\{.3,.4,.5,.6\}$, $\sigma\in\{.08,.12,.16,.20\}$.
  \item \textbf{Seeding}: random, central, marginal, closeness, eigenvector.
\end{itemize}
\vspace{0.2cm}
\begin{center}
Total: $\;41\times13\times4\times4\times5\times \text{(runs)} \;\approx\; \emphr{$4\times10^{6}$ simulations}$
\end{center}
\vspace{0.2cm}
We then fit \emph{Big Additive Models} (BAM) to isolate the structural effect:
\begin{equation*}
  \mathrm{logit}\big[\mathbb{E}(\Phi)\big]=\beta_0+\mathbf{Z}\boldsymbol\beta
  +\beta_{\text{DP}}\,\mathbb{I}_{\text{DP}}+s(\mathrm{IUL},\mathrm{MSP})
\end{equation*}
\end{frame}

\sectionslide{Conclusions}

% ---- Conclusions structured as answers to the two opening questions (point 18 + 16)
\begin{frame}{Conclusions}

\vspace{0.25cm}
\textbf{\color{black} Can the shape of realistic networks \emph{alone} change the outcome?}
\vspace{0.10cm}
\begin{itemize}\small
  \item \emphr{Volume:} empirical homophily yields a \emphr{structural premium} ($+57\%$ odds).
  \item When society is not too restrictive (high MSP), \emphr{highly attractive and unattractive
        innovations reach nearly the same adoption} — purely through social influence.
\end{itemize}
\vspace{0.25cm}
\textbf{\color{black} Is structure an autonomous force deciding \emph{how} it spreads?}
\vspace{0.10cm}
\begin{itemize}\small
  \item \emphr{Dynamics:} in realistic networks the transition is \emphr{continuous} ($\gamma\approx1$) —
        across comparable societies, tipping points carry \emph{statistical} early warning
        (rising variance, slow recovery); randomization erases it (\emphr{$\gamma\approx4$}, abrupt).
  \item This signposted regime is \emphr{exclusive} to empirically homophilic structure.
\end{itemize}
\vspace{0.2cm}
\begin{center}\small
\emph{Macroscopic social change is not the linear sum of individual preferences —
it is governed by the structure of our ties}.
\end{center}
\end{frame}

\begin{frame}{Appendix — The mean-field paradox \& its resolution}
\begin{block}{\small Puzzle}
\small Mean-field $\gamma=1$ usually needs a \emph{well-mixed} system.
But we have a \emph{strongly clustered, homophilic} network.
\end{block}
\vspace{0.1cm}
\begin{block}{\small Resolution}
\footnotesize Complex networks have no finite spatial dimension — neighborhoods expand
exponentially, so the system sits \emph{above} the upper critical dimension and mean-field
is the \emph{generic} expectation. The real contrast is the \emph{order} of the transition:
homophilic clustering yields a continuous (2nd-order) transition, whereas degree-preserving
randomization collapses into a discontinuous (1st-order) regime shift.
\end{block}
\vspace{0.1cm}
\begin{center}\small
Homophily gives the system a \emphr{``sociological viscosity''}: across comparable societies,
tipping points carry \emph{statistical} early warning (rising variance, slow recovery)
instead of detonating blind.
\end{center}
\end{frame}
