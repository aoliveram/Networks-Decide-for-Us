% SLIDE 2: MOTIVATION
% ============================================================================
\begin{frame}{Why Networks Matter for Adoption}

\begin{itemize}
  \item Collective behaviors (innovations, social movements, beliefs) spread through networks.

  \vspace{0.5cm}
  \item Classical models: homogeneous populations, synthetic topologies.

  \vspace{0.5cm}
  \item Reality: people cluster by similarity (\textcolor{accent_orange}{homophily}).

  \vspace{0.5cm}
  \item \textbf{Question:} Does empirical network structure act as an autonomous causal force?

  \vspace{0.5cm}
  \item \textbf{Hypothesis:} Homophilic clustering governs both the \textit{volume} and \textit{dynamics} of collective change.
\end{itemize}

\end{frame}

% ============================================================================
% SLIDE 3: PHYSICS ANALOGY
% ============================================================================
\begin{frame}{From Physics to Society: The Mapping}

\begin{columns}
  \column{0.5\textwidth}
  \textbf{Ferromagnetic System (Ising)}
  \begin{itemize}
    \item Spins: align or anti-align
    \item Temperature $T$: disorder
    \item Field $H$: external bias
    \item Order parameter: Magnetization $M$
  \end{itemize}

  \column{0.5\textwidth}
  \textbf{Social Adoption}
  \begin{itemize}
    \item Agents: adopt or abstain
    \item \textcolor{accent_orange}{MSP} (Max Social Proximity): social flexibility
    \item \textcolor{accent_orange}{IUL} (Intrinsic Utility): innovation attractiveness
    \item Order parameter: Adoption $\Phi$
  \end{itemize}
\end{columns}

\vspace{1cm}

\centering
\boxed{\text{Low MSP} = \text{rigid clusters (frozen)} \quad | \quad \text{High MSP} = \text{flexible (melted)}}

\end{frame}

\begin{frame}{Methods: From Data to Simulation}

  \begin{itemize}
    \item Homophily parameters from GSS core discussion networks (Smith et al., 2014)
    \item ERGM simulation onto ATP 2014 respondents ($N=1000$)
  \end{itemize}

  \vspace{0.4cm}
  \item \textbf{Agent-Based Model (netdiffuseR)}
  \begin{itemize}
    \item Hybrid: Rational adoption (q_i ≤ Γ) + Social influence (selective by similarity)
    \item ~4 million simulations sweeping IUL × MSP × threshold parameters
  \end{itemize}

  \vspace{0.4cm}
  \item \textbf{Analysis}
  \begin{itemize}
    \item Generalized Additive Models (GAM): measure structural premium
    \item Phase transition analysis: extract critical exponents (γ)
  \end{itemize}
\end{enumerate}

\end{frame}

% SLIDE 5: RESULT 1 - STRUCTURAL PREMIUM
% ============================================================================
\begin{frame}{Result 1: The Structural Premium}

\vspace{-0.3cm}
\begin{itemize}
  \item \textbf{Finding:} Randomizing network structure $\Rightarrow$ 57\% drop in adoption.

  \vspace{0.3cm}
  \item \textbf{Interpretation:} Homophilic clustering acts as a structural \textit{catalyst}, not just a passive conduit.

  \vspace{0.3cm}
  \item Local echo chambers allow social reinforcement, even when population is averse to innovation.

  \begin{center}
    \vspace{0.3cm}
    \boxed{\text{β}_{\text{structure}} = -0.850 \quad (p < 0.001)}
  \end{center}

\end{itemize}

\vspace{0.5cm}

\centering
\textit{Volume: Networks amplify diffusion. But do they also shape \textbf{how} it happens?}

\end{frame}

% ============================================================================
% SLIDE 6: RESULT 2 - CRITICAL EXPONENTS
% ============================================================================
\begin{frame}{Result 2: Empirical Criticality}

\vspace{-0.3cm}

In empirical networks:
\begin{itemize}
  \item \textbf{Mean-Field universality}: $\gamma \approx 0.977 \approx 1$
  \item Susceptibility diverges as power law: $\chi \sim |MSP - MSP_c|^{-\gamma}$
  \item Critical Slowing Down observed
  \item \textcolor{accent_orange}{Continuous phase transition (2nd order)}
\end{itemize}

\vspace{0.3cm}

In randomized networks:
\begin{itemize}
  \item \textbf{No universality}: $\gamma \approx 4.0$ (unphysical)
  \item Power-law breaks down
  \item Abrupt regime shifts
  \item \textcolor{accent_orange}{Discontinuous transition (1st order)}
\end{itemize}

\vspace{0.4cm}

\centering
\boxed{\text{Separation ratio} = \frac{\gamma_{\text{random}}}{\gamma_{\text{empirical}}} \approx 4.3\times}

\end{frame}

% ============================================================================
% SLIDE 7: THE MEAN-FIELD PARADOX
% ============================================================================
\begin{frame}{The Paradox: Why Mean-Field in Clustered Networks?}

\vspace{-0.3cm}

\begin{block}{The Puzzle}
Mean-Field universality ($\gamma = 1$) typically requires well-mixed systems.\\
But we observe it in highly clustered, homophilic networks.
\end{block}

\vspace{0.3cm}

\begin{block}{The Resolution}
Homophilic clusters act as quasi-isolated \textit{subsystems}.\\
At criticality (MSP $\approx$ MSP$_c$), weak inter-cluster bridges suddenly activate.\\
This global synchronization is indistinguishable from mean-field behavior.
\end{block}

\vspace{0.3cm}

\centering
\textit{Homophily stabilizes local order \& enables global coherence simultaneously.}

\end{frame}
