% ---------------------------------------------------------------- OVERVIEW
\begin{frame}{The question}
\begin{itemize}
  \item Collective behaviors (innovations, mobilizations) spread as \emphr{adoption cascades} on social networks.
  \vspace{0.3cm}
  \item Most models use \emph{synthetic} topologies and treat agents as cognition-free relays \refn{1}.
  \vspace{0.3cm}
  \item Real networks are shaped by \emphr{multidimensional homophily} — similar people cluster \refn{2}.
\end{itemize}
\vspace{0.5cm}
\begin{center}
\fcolorbox{ndred}{white}{\parbox{0.86\textwidth}{\centering
Conditional on \emph{fixed} individual propensities, does empirical homophilic
structure act as an \emphr{autonomous causal force} — shaping not just \emph{how much}
adoption occurs, but the \emphr{physics of its tipping points}?}}
\end{center}
\end{frame}

% ================================================================ THE MODEL
\sectionslide{The model}

% ---- Rational choice
\begin{frame}{The model — 1) Rational Choice}
An innovation has an \emph{Intrinsic Utility Level} (IUL) $\Gamma\in[0,1]$.
Each agent has a \emph{Minimum Utility Requirement} (MUR) $q_i\in[0,1]$ (its aversion).
\vspace{0.3cm}
\begin{center}
An agent adopts on its own iff
\qquad \fcolorbox{ndred}{white}{$\;q_i \le \Gamma \;\implies\; a_i = 1\;$}
\end{center}
\vspace{0.4cm}
\begin{center}
\begin{tikzpicture}[every node/.style={circle,draw,minimum size=6mm,inner sep=0pt},
                    adopt/.style={fill=black,text=white}, scale=1.0]
  \node[adopt] (1) at (0,1.2) {}; \node (3) at (2.2,1.2) {};
  \node (2) at (-1.1,0) {}; \node (4) at (1.1,0) {}; \node (6) at (2.6,-0.2) {};
  \node (5) at (0.2,-1.2) {};
  \draw (1)--(4) (2)--(4) (4)--(6) (4)--(5) (1)--(2);
  \node[ndgray,draw=none] at (-1.9,0) {\scriptsize $q_2<\Gamma$};
  \node[ndgray,draw=none] at (1.95,0.35) {\scriptsize $q_4<\Gamma$};
  \node[ndgray,draw=none] at (-0.6,-1.5) {\scriptsize $q_5>\Gamma$};
\end{tikzpicture}
\end{center}
\begin{center}\footnotesize Adoption by intrinsic utility is \emphr{deterministic} and \emphr{topology-blind}.\end{center}
\end{frame}

% ---- Selective social influence
\begin{frame}{The model — 2) Selective Social Influence}
If intrinsic adoption fails, agents may adopt by peer contagion \refn{5,6}. Effective exposure:
\begin{equation*}
  \tilde{E}_i \equiv \frac{\sum_{j\neq i}\mathbf{X}_{ij}\,\tilde{a}_j}{\sum_{j\neq i}\mathbf{X}_{ij}},
  \qquad a_i=\begin{cases}1 & \text{if } \tau_i \le \tilde{E}_i\\[2pt]0 & \text{otherwise}\end{cases}
\end{equation*}
Only \emph{influential} infected peers count — those socially \emph{close} in Blau space:
\begin{equation*}
  \tilde{a}_j = 1 \iff a_j = 1 \;\wedge\; U_{ij} \le \sigma\!\left(MSP - d_{ij}\right),
  \qquad U_{ij}\sim\mathrm{Unif}(0,1)
\end{equation*}
\vspace{-0.1cm}
\begin{itemize}
  \item $d_{ij}$: social distance between $i$ and $j$; \quad $\sigma(z)=1/(1+e^{-z/T})$.
  \item \emphr{MSP} (Maximum Social Proximity): how far across demographic lines influence can reach. Socially similar peers persuade more \refn{7}.
\end{itemize}
\begin{center}\footnotesize Social adoption is \emphr{stochastic} (via $U_{ij}$) and \emphr{topology-sensitive}.\end{center}
\end{frame}

% ================================================================ SETTING UP
\sectionslide{Imputing structure \& running the model}

\begin{frame}{Imputing a network onto a survey \refn{3}}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{General Social Survey (GSS 2004)}
\begin{itemize}\footnotesize
  \item Representative; demographics (age, sex, educ, race, religion); \emph{ego} network module.
\end{itemize}
\textbf{American Trends Panel (ATP 2014)}
\begin{itemize}\footnotesize
  \item Representative; same demographics; \emph{no} network — but rich propensity items ($q_i$).
\end{itemize}
\vspace{0.2cm}
Homophily strengths $\hat{\boldsymbol\beta}$ from GSS confidant data \refn{3,4} feed an \emphr{ERGM}:
\begin{equation*}\footnotesize
  \mathrm{logit}\,\Pr(Y_{ij}=1)=\alpha+\boldsymbol\beta^\top \mathbf{d}_{ij}
\end{equation*}
$\Rightarrow$ plausible discrete topologies ($N=1000$) on ATP respondents.
\column{0.45\textwidth}
\centering
\includegraphics[width=\linewidth]{../../plots/02_GSS_network_ergm_tests/GSS_degree_dist.pdf}\\
{\scriptsize Right-skewed degree structure that ER / BA models fail to match jointly.}
\end{columns}
\end{frame}

\begin{frame}{Comprehensive simulation design}
Hybrid social contagion via \texttt{netdiffuseR} \refn{9}, over plausible (GSS) vs.\
\emphr{degree-preserving randomized} (GSS-DP) topologies.
\vspace{0.2cm}
\begin{itemize}
  \item \textbf{IUL} ($\Gamma$): 41 levels $[0,1]$. \quad \textbf{MSP} ($h$): 13 levels $[0,1]$.
  \item \textbf{Thresholds} $\tau_i\sim\mathcal{N}(\mu,\sigma)$: $\mu\in\{.3,.4,.5,.6\}$, $\sigma\in\{.08,.12,.16,.20\}$.
  \item \textbf{Seeding}: random, central, marginal, closeness, eigenvector.
\end{itemize}
\vspace{0.2cm}
\begin{center}
Total: $\;41\times13\times4\times4\times5\times \text{(runs)} \;\approx\; \emphr{$4\times10^{6}$ simulations}$
\end{center}
\vspace{0.2cm}
We then fit \emph{Big Additive Models} (BAM) to isolate the structural effect:
\begin{equation*}\footnotesize
  \mathrm{logit}\big[\mathbb{E}(\Phi)\big]=\beta_0+\mathbf{Z}\boldsymbol\beta
  +\beta_{\text{DP}}\,\mathbb{I}_{\text{DP}}+s(\mathrm{IUL},\mathrm{MSP})
\end{equation*}
\end{frame}

\begin{frame}{Mapping social adoption to a ferromagnet}
\begin{columns}[c]
\column{0.56\textwidth}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}l@{\;\;}c@{\;\;}l@{}}
\toprule

\midrule

Magnetization $M$           & $\leftrightarrow$ & Adoption $\Phi=N_{\text{ad}}/N$ \\
External field $H$          & $\leftrightarrow$ & \emphr{IUL} $\Gamma$ (rational bias) \\
Temperature $T$             & $\leftrightarrow$ & \emphr{MSP} $h$ (social agitation) \\
Exchange $J_{ij}$           & $\leftrightarrow$ & Homophily / clustering \\
Susceptibility $\chi$       & $\leftrightarrow$ & $\mathrm{Var}(\Phi)$ (FDT) \\
\bottomrule
\end{tabular}
\column{0.44\textwidth}
\footnotesize
Pseudo-Hamiltonian:
\begin{equation*}
\mathcal{H}\approx -\!\!\sum_{\langle i,j\rangle}\! w_{ij}(h)\,s_i s_j \;-\; \Gamma\sum_i s_i
\end{equation*}
\vspace{0.1cm}
\emphr{Low MSP} $\to$ rigid echo chambers (frozen domains).\\[3pt]
\emphr{High MSP} $\to$ boundaries ``melt'', global cascades possible.
\end{columns}
\vspace{0.3cm}
\begin{center}\footnotesize
Susceptibility diverges near criticality \refn{8}: $\;\chi=\mathrm{Var}(\Phi)\propto |MSP-MSP_c|^{-\gamma}$
\end{center}
\end{frame}

\begin{frame}{Why only the MSP axis discriminates}
\begin{columns}[c]
\column{0.52\textwidth}
\textbf{Two control parameters, two channels:}
\begin{itemize}\footnotesize
  \item \emphr{Path 2 (MSP, $\gamma$)}: drives the \emph{social} channel.
        $\gamma_{\text{GSS}}\!\approx\!1$ vs.\ $\gamma_{\text{DP}}\!\approx\!4$ — \emphr{topology-sensitive}.
  \item \emphr{Path 1 (IUL, $\delta$)}: drives the \emph{rational} channel ($q_i\le\Gamma$ is node-local).
        $\delta_{\text{GSS}}\!\approx\!2.85$, $\delta_{\text{DP}}\!\approx\!2.91$ — \emphr{topology-blind} (both $\approx$ MF $\delta=3$).
\end{itemize}
\vspace{0.2cm}
\footnotesize The field response reflects the \emph{MUR distribution}, shared by both topologies.
The universality \emphr{discriminator} must therefore be the MSP (temperature) axis.

% ================================================================ DISCUSSION
\begin{frame}{The mean-field paradox — and its resolution}
\begin{block}{\footnotesize Puzzle}
\footnotesize Mean-field $\gamma=1$ usually needs a \emph{well-mixed} system.
We recover it in a \emph{strongly clustered, homophilic} network.
\end{block}
\vspace{0.1cm}
\begin{block}{\footnotesize Resolution}
\footnotesize Homophilic clusters behave as quasi-isolated subsystems coupled by
\emph{weak, similarity-selective bridges}. As MSP $\to MSP_c$ these bridges activate
together, synchronizing clusters into one coherent order parameter — \emph{indistinguishable}
from mean-field behavior. Randomization (DP) destroys this and yields discontinuity instead.
\end{block}
\vspace{0.1cm}
\begin{center}\footnotesize
Homophily provides a \emphr{``sociological viscosity''} that converts chaotic jumps into legible cascades.
\end{center}
\end{frame}

% ================================================================ CONCLUSION
\sectionslide{Conclusions}

\begin{frame}{Conclusions}
\begin{center}{\large \emphr{Networks decide for us} — twice over.}\end{center}
\vspace{0.4cm}
\begin{enumerate}
  \item \textbf{Volume:} empirical homophily yields a \emphr{structural premium} (+57\% odds).
  \vspace{0.2cm}
  \item \textbf{Dynamics:} homophily makes tipping points \emphr{continuous \& predictable}
        (mean-field, $\gamma\approx1$); randomization makes them \emphr{abrupt \& volatile}
        ($\gamma\approx4$).
  \vspace{0.2cm}
  \item \textbf{Mechanism:} the social (MSP) axis carries universality; the rational (IUL)
        axis is topology-blind ($\delta\approx3$).
\end{enumerate}
\vspace{0.4cm}
\begin{center}\footnotesize
Macroscopic social change is not the linear sum of individual preferences —
it is governed by the structure of our ties.
\end{center}
\end{frame}
