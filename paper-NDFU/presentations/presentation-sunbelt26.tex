% ---------------------------------------------------------------- 2. MOTIVATION
\begin{frame}{Collective behavior spreads on networks}
\begin{itemize}
  \item Collective behaviors — innovations, mobilizations — spread as
        \emphr{contagions} on social networks.
  \vspace{0.4cm}
  \item Real human networks have a specific texture: we cluster with people
        like us — \emphr{multidimensional homophily} \refn{1}.
  \vspace{0.4cm}
  \item So a natural question: how much of \emph{where} a behavior ends up is about
        \emph{who people are}, and how much is about \emph{how they are wired together}?
\end{itemize}
\end{frame}

% ---------------------------------------------------------------- 3. THE GAP
\begin{frame}{The gap}
\begin{itemize}
  \item Diffusion ABMs run on \emph{made-up} networks (small-world, scale-free) \refn{2}.
  \vspace{0.25cm}
  \item Rich surveys (ATP, GSS) measure \emph{propensities} but carry \emph{no} relational data.
  \vspace{0.25cm}
  \item Threshold models often assume \emph{all} adoption needs peer influence — and when
        weak-interest groups still tip, the usual explanation is \emph{stochastic noise} that
        spontaneously instigates a cascade \refn{8}.
\end{itemize}
\vspace{0.4cm}
\begin{center}\small
We ask whether \emphr{measured preference and real structure} — not noise, not synthetic
graphs — can do that work.
\end{center}
\end{frame}

% ---------------------------------------------------------------- 3b. WHERE WE GO
\begin{frame}{Where we go}
\textbf{Two moves}
\begin{itemize}
  \item \emphr{Impute} plausible whole-networks onto real survey respondents from
        empirical homophily — going from McPherson \& Smith's latent ``context'' term \refn{4}
        to \emphr{full simulable networks}.
  \vspace{0.2cm}
  \item \emphr{Measure} the causal premium of that structure on diffusion — and ask
        \emph{what about the structure} actually matters.
\end{itemize}
\vspace{0.4cm}
\begin{center}
\fcolorbox{ndred}{white}{\parbox{0.92\textwidth}{\centering
Holding individual propensities \emph{fixed}, does the \emphr{shape of the network alone} —
not who its members are — decide \emph{how much} collective behavior spreads?}}
\end{center}
\end{frame}

% ================================================================ THE MODEL
\sectionslide{The model}

% ---- 4. Rational choice
\begin{frame}{Adoption, part 1 — Rational Choice}
\begin{itemize}
  \item A behavior has an \emph{Intrinsic Utility Level} \emphr{IUL}, $\Gamma\in[0,1]$.
  \item Each person has a \emph{Minimum Utility Requirement} \emphr{MUR}, $q_i\in[0,1]$,
        representing their aversion to adopting the behavior.
\end{itemize}
\vspace{0.1cm}
\begin{center}
An agent adopts on its own iff\quad
\fcolorbox{ndred}{white}{$\;q_i \le \Gamma \;\implies\; a_i = 1\;$}
\end{center}
\vspace{0.15cm}
\begin{center}
\includegraphics[width=0.95\textwidth]{graph-rational-choice.png}
\end{center}
\begin{center}\footnotesize
This channel is \emphr{topology-blind} — it does not need the network at all.
\end{center}
\end{frame}

% ---- 4. Selective social influence
\begin{frame}{Adoption, part 2 — Selective Social Influence}
Otherwise, an agent adopts by peer contagion if $\tau_i \le \tilde{E}_i$ \refn{3}
— but only peers who are socially \emph{close} \refn{6} count:
\vspace{0.05cm}
\begin{equation*}
  \tilde{E}_i \equiv \frac{\sum_{j\neq i}\mathbf{X}_{ij}\,\tilde{a}_j}{\sum_{j\neq i}\mathbf{X}_{ij}},
  \qquad
  \tilde{a}_j = 1 \iff a_j = 1 \;\wedge\; U_{ij} \le \sigma\!\left(MSP - d_{ij}\right)
\end{equation*}
\begin{itemize}\small
  \item $d_{ij}$: social (Blau-space) distance; \emphr{MSP} (Maximum Social Proximity)
        sets how far across demographic lines influence can reach.
\end{itemize}
\vspace{0.05cm}
\begin{center}
\includegraphics[width=0.9\textwidth]{graph-social-influence.png}
\end{center}
\end{frame}

% ---- 6c. Simulation design (copied from CPIN deck) ----
\begin{frame}{Comprehensive simulation design}
Hybrid social contagion via \texttt{netdiffuseR} \refn{7}, over plausible \emphr{GSS} vs.\
degree-preserving randomized \emphr{GSS-DP} topologies.
\vspace{0.2cm}
\begin{itemize}
  \item \textbf{IUL} ($\Gamma$): 41 levels $[0,1]$. \quad \textbf{MSP} ($h$): 13 levels $[0,1]$.
  \item \textbf{Thresholds} $\tau_i\sim\mathcal{N}(\mu,\sigma)$: $\mu\in\{.3,.4,.5,.6\}$, $\sigma\in\{.08,.12,.16,.20\}$.
  \item \textbf{Seeding}: random, central, marginal, closeness, eigenvector.
\end{itemize}
\vspace{0.2cm}
\begin{center}
Total: $\;41\times13\times4\times4\times5\times \text{(runs)} \;\approx\; \emphr{$4\times10^{6}$ simulations}$
\end{center}
\vspace{0.2cm}
We then fit \emph{Big Additive Models} (BAM) to isolate the structural effect:
\begin{equation*}
  \mathrm{logit}\big[\mathbb{E}(\Phi)\big]=\beta_0+\mathbf{Z}\boldsymbol\beta
  +\beta_{\text{DP}}\,\mathbb{I}_{\text{DP}}+s(\mathrm{IUL},\mathrm{MSP})
\end{equation*}
\end{frame}

% ---- 7. RESULT 1: structural premium ----
\begin{frame}{Result 1 — A structural premium}
\begin{columns}[c]
\column{0.46\textwidth}

\vspace{0.2cm}
\begin{center}

% ================================================================ CLOSE
% ---- conclusions ----
\begin{frame}{Conclusions}
\begin{center}{\large \emphr{Networks decide for us.}}\end{center}
\vspace{0.25cm}
\begin{enumerate}
  \item \textbf{A method:} impute \emphr{simulable whole-networks} onto survey respondents
        from empirical homophily — reusable wherever you have propensities but no ties.
  \vspace{0.1cm}
  \item \textbf{A finding:} holding propensities fixed, empirical structure yields a
        \emphr{$+57\%$ adoption premium}.
  \vspace{0.1cm}
  \item \textbf{A mechanism:} what ``breaking structure'' destroys is, above all,
        \emphr{homophily} — neighbors stop being similar.
\end{enumerate}
\vspace{0.2cm}
\begin{center}\small
Unlike Macy \& Evtushenko \refn{8}, the idiosyncrasy that drives this is
\emphr{measured preference}, not stochastic noise: \emph{the same agents, on a different
topology, produce a different society.}
\end{center}
\end{frame}

\begin{frame}{Backup — the \emph{shape} of change also differs}
\begin{columns}[c]
\column{0.5\textwidth}
\small
Beyond \emph{how much}, structure governs \emph{how} adoption tips:
\begin{itemize}\small
  \item Plausible (GSS) $\rightarrow$ \emphr{continuous, predictable} transition.
  \item Randomized (GSS-DP) $\rightarrow$ \emphr{abrupt} regime shift.
\end{itemize}
\vspace{0.2cm}

ones do not \refn{9}.
