% ---------------------------------------------------------------- 2. THE QUESTION
\begin{frame}{The question}
\begin{itemize}
  \item Collective behaviors — innovations, mobilizations — spread as
        \emphr{contagions} on social networks.
  \vspace{0.3cm}
  \item Real human networks have a specific texture: we cluster with people
        like us — \emphr{multidimensional homophily} \refn{1}.
  \vspace{0.3cm}
  \item Yet most diffusion models run on \emph{synthetic} topologies, and most
        surveys give us \emph{individuals} but \emph{not their networks}.
\end{itemize}
\vspace{0.5cm}
\begin{center}
\fcolorbox{ndred}{white}{\parbox{0.92\textwidth}{\centering
Holding individual propensities \emph{fixed}, does the \emphr{shape of the network
alone} — not who its members are — decide \emph{how much} collective behavior spreads?}}
\end{center}
\end{frame}

% ---------------------------------------------------------------- 3. THE GAP
\begin{frame}{The gap, and where we go}
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{The standing problems}
\begin{itemize}\small
  \item Diffusion ABMs use \emph{made-up} networks (small-world, scale-free) \refn{2}.
  \item Rich surveys (ATP, GSS) measure \emph{propensities} but have \emph{no} relational data.
  \item Threshold models often assume \emph{all} adoption needs peer influence \refn{3}.
\end{itemize}
\column{0.48\textwidth}
\textbf{Our two moves}
\begin{itemize}\small
  \item \emphr{Impute} plausible whole-networks onto real survey respondents
        from empirical homophily.
  \item \emphr{Measure} the causal premium of that structure on diffusion,
        and ask \emph{what about it} matters.
\end{itemize}
\end{columns}
\vspace{0.5cm}
\begin{center}\small
Building on \emph{imputing social context from survey data} \refn{4} —
but going from a latent context term to \emphr{full simulable networks}.
\end{center}
\end{frame}

% ================================================================ THE MODEL
\sectionslide{The model}

% ---- 4. Rational choice
\begin{frame}{Adoption, part 1 — Rational Choice}
\begin{itemize}
  \item A behavior has an \emph{Intrinsic Utility Level} (IUL) $\Gamma\in[0,1]$ —
        a latent quality scale: the highest-IUL option is the one that wins the
        most \emphr{pairwise comparisons} \refn{5}.
  \item Each person has a \emph{Minimum Utility Requirement} $q_i\in[0,1]$
        (their aversion).
\end{itemize}
\vspace{0.1cm}
\begin{center}
An agent adopts on its own iff\quad
\fcolorbox{ndred}{white}{$\;q_i \le \Gamma \;\implies\; a_i = 1\;$}
\end{center}
\vspace{0.15cm}
\begin{center}
\includegraphics[width=0.95\textwidth]{graph-rational-choice.png}
\end{center}
\begin{center}\footnotesize
This channel is \emphr{topology-blind} — it does not need the network at all.
\end{center}
\end{frame}

% ---- 4. Selective social influence
\begin{frame}{Adoption, part 2 — Selective Social Influence}
Otherwise, an agent adopts by peer contagion if $\tau_i \le \tilde{E}_i$ \refn{3}
— but only peers who are socially \emph{close} \refn{6} count:
\vspace{0.05cm}
\begin{equation*}
  \tilde{E}_i \equiv \frac{\sum_{j\neq i}\mathbf{X}_{ij}\,\tilde{a}_j}{\sum_{j\neq i}\mathbf{X}_{ij}},
  \qquad
  \tilde{a}_j = 1 \iff a_j = 1 \;\wedge\; U_{ij} \le \sigma\!\left(MSP - d_{ij}\right)
\end{equation*}
\begin{itemize}\small
  \item $d_{ij}$: social (Blau-space) distance; \emphr{MSP} (Maximum Social Proximity)
        sets how far across demographic lines influence can reach.
\end{itemize}
\vspace{0.05cm}
\begin{center}
\includegraphics[width=0.9\textwidth]{graph-social-influence.png}
\end{center}
\end{frame}

% ---- 6. Simulation design ----
\begin{frame}{Simulation design}
Hybrid social contagion via \texttt{netdiffuseR} \refn{7}, over plausible (GSS) vs.\
\emphr{degree-preserving randomized} (GSS-DP) topologies.
\vspace{0.25cm}
\begin{itemize}
  \item \textbf{IUL} ($\Gamma$): 41 levels. \quad \textbf{MSP} ($h$): 13 levels.
  \item \textbf{Thresholds} $\tau_i\sim\mathcal{N}(\mu,\sigma)$; \textbf{seeding}: random, central, marginal, closeness, eigenvector.
\end{itemize}
\vspace{0.25cm}
\begin{center}
$\;41\times13\times4\times4\times5\times \text{(runs)}\;\approx\;\emphr{$4\times10^{6}$ simulations}$
\end{center}
\vspace{0.25cm}
The DP baseline keeps each person's \emph{number} of ties but rewires \emph{whom}
they connect to — our counterfactual for ``the same people, different structure.''
\end{frame}

% ================================================================ RESULTS
\sectionslide{Results}

% ---- 7. RESULT 1: structural premium ----
\begin{frame}{Result 1 — A structural premium}
\begin{columns}[c]
\column{0.46\textwidth}
A Big Additive Model isolates the marginal effect of destroying empirical structure:
\vspace{0.2cm}
\begin{center}
\fcolorbox{ndred}{white}{$\;\beta_{\text{DP}}=-0.850^{***}\;$ (OR $=0.43$)}\\[4pt]
{\small $\rightarrow$ randomization cuts adoption odds by $\sim$\emphr{57\%}}
\end{center}
\vspace{0.2cm}
\small Same people, same propensities — only the \emphr{wiring} changed.
Homophilic clustering lets social reinforcement build inside local echo chambers.
\column{0.54\textwidth}
\centering

% ---- 9. literature connection ----
\begin{frame}{Where this sits}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Macy \& Evtushenko (2020)} \refn{8}
\begin{itemize}\small
  \item Reintroduce individual idiosyncrasy as \emph{stochastic noise}.
  \item Noise can \emph{spontaneously instigate} cascades a weak-interest group would otherwise doom.
\end{itemize}
\column{0.5\textwidth}
\textbf{Here}
\begin{itemize}\small
  \item Idiosyncrasy is \emphr{measured preference}, not noise.
  \item At low IUL, \emph{rational} adopters are endogenous instigators — we go one
        level \emph{behind} spontaneous instigation.
  \item And \emphr{structure}, not just the threshold distribution, governs the outcome.
\end{itemize}
\end{columns}
\vspace{0.5cm}
\begin{center}\small
\emph{The same agents, on a different topology, produce a different society.}
\end{center}
\end{frame}

% ---- 10. teaser: the SHAPE of change (single physics slice) ----
\begin{frame}{Teaser — the \emph{shape} of change also differs}
\begin{columns}[c]
\column{0.54\textwidth}
\small
Beyond \emph{how much}, structure governs \emph{how} adoption tips:
\begin{itemize}\small
  \item Plausible networks $\rightarrow$ a \emphr{continuous, predictable} transition.
  \item Randomized networks $\rightarrow$ an \emphr{abrupt, all-or-nothing} regime shift.
\end{itemize}
\vspace{0.2cm}
There is even a signature of \emphr{universality} (mean-field scaling) — a story for
another talk \refn{9}. Happy to discuss in Q\&A.

\end{columns}
\end{frame}

% ================================================================ CLOSE
% ---- 11. conclusions ----
\begin{frame}{Conclusions}
\begin{center}{\large \emphr{Networks decide for us.}}\end{center}
\vspace{0.3cm}
\begin{enumerate}
  \item \textbf{A method:} impute \emphr{simulable whole-networks} onto survey respondents
        from empirical homophily — reusable wherever you have propensities but no ties.
  \vspace{0.15cm}
  \item \textbf{A finding:} holding propensities fixed, empirical structure yields a
        \emphr{$+57\%$ adoption premium}.
  \vspace{0.15cm}
  \item \textbf{A mechanism:} what ``breaking structure'' destroys is, above all,
        \emphr{homophily} — neighbors stop being similar.
\end{enumerate}
\vspace{0.3cm}
\begin{center}\small
\emph{Macroscopic diffusion is not the sum of individual preferences —
it is governed by the structure of our ties.}
\end{center}
\end{frame}

% ---- 12. limitations / next ----
\begin{frame}{Limitations \& next steps}
\begin{itemize}
  \item \emphr{Separate homophily from clustering:} the attribute-shuffle null model —
        identical topology, permuted node attributes — isolates pure homophily.
  \vspace{0.2cm}
  \item \emphr{Heterogeneous MSP:} social porosity that varies by group, not a global constant.
  \vspace{0.2cm}
  \item \emphr{Higher-stakes behavior:} extend the propensity scores to risky / high-cost
        collective action.
  \vspace{0.2cm}
  \item \emphr{The criticality story:} the continuous-vs-abrupt transition and its
        universality (the teaser) — a companion paper.
\end{itemize}
\end{frame}
